cmake_minimum_required(VERSION 3.22)
project(minifire_zygisk LANGUAGES CXX)
add_library(minifire_zygisk SHARED src/zygisk_entry.cpp)
add_library(minifire_core SHARED src/native_loader.cpp src/native_entry.cpp src/login_filter.cpp src/binary_guard.cpp)
foreach(target minifire_zygisk minifire_core)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fvisibility=hidden -fstack-protector-strong)
  target_link_options(${target} PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,relro -Wl,-z,now)
endforeach()
target_compile_options(minifire_zygisk PRIVATE -fno-exceptions -fno-rtti)
target_link_options(minifire_zygisk PRIVATE -nostdlib++)
target_link_libraries(minifire_zygisk PRIVATE stdc++ dl log)
target_link_libraries(minifire_core PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/libdobby.a dl log)
